## [1. 两数之和](https://leetcode.cn/problems/two-sum/)

```
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        unordered_map<int, int> h;

        for (int i = 0; i < nums.size(); i++) {
            int other = target - nums[i];

            if (h.count(other)) {
                return {i, h[other]};
            }

            h[nums[i]] = i;
        }

        return {};
    }
};
```

```
class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        unordered_set<int> s(nums.begin(), nums.end());

        int res = 0;

        for (auto num : s) {
            if (!s.count(num - 1)) {
                int curlen = 1;
                int curnum = num;

                while (s.count(curnum + 1)) {
                    curlen++;
                    curnum++;
                }

                res = max(res, curlen);
            }
        }

        return res;
    }

## [15. 三数之和](https://leetcode.cn/problems/3sum/)

**注意：本道题相同的第一个数只处理一次**

如：[-1, -1, 0, 1, 2]，答案不能有两个[-1, 0, 1]

```
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        vector<vector<int>> res;

        sort(nums.begin(), nums.end());

        for(int i = 0; i < nums.size(); i ++ ){
            if(nums[i] > 0) break;

            if(i > 0 && nums[i] == nums[i - 1]) continue;

            int l = i + 1;
            int r = nums.size() - 1;

            while(l < r){
                int t = nums[i] + nums[l] + nums[r];

                if(t == 0){
                    res.push_back({nums[i], nums[l], nums[r]});

                    while(l < r && nums[l] == nums[l + 1]) l ++;
                    while(l < r && nums[r] == nums[r - 1]) r --;

                    l ++, r --;
                }
                else if(t < 0) l ++;
                else r --;
            }
        }
        return res;
    }
};
```

## [42. 接雨水](https://leetcode.cn/problems/trapping-rain-water/)

核心代码：

```
leftMax[i]   ：从左边到当前位置，最高高度
rightMax[i]  ：从当前位置到右边，最高高度
二者较小值   ：水面最高能达到的高度
height[i]    ：当前位置柱子本身占据的高度
```

所以：

```
当前位置的水量 = 水面高度 - 柱子高度
```

```
class Solution {
public:
    int trap(vector<int>& height) {
        int n = height.size();

        vector<int> lmax(n);
        vector<int> rmax(n);

        lmax[0] = height[0];
        for(int i = 1; i < n; i ++ ) lmax[i] = max(lmax[i - 1], height[i]);

        rmax[n - 1] = height[n - 1];
        for(int i = n - 2; i >= 0; i -- ) rmax[i] = max(rmax[i + 1], height[i]);

        int res = 0;
        for(int i = 0; i < n; i ++ ){
            res += min(lmax[i], rmax[i]) - height[i];
        }

        return res;
    }
};
```

            return ans;
        }

                ans.push_back(l);
            }

            w[s[l] - 'a'] -- ;//每次右移最左边的数出窗口
        }

        return ans;
    }
};

## [239. 滑动窗口最大值](https://leetcode.cn/problems/sliding-window-maximum/)

这道题主要是需要搞清楚，队头和队尾元素是怎么定义的，定义好了之后如何判空

本题：

> hh = 0, tt = -1
>
> 判断为空就看 hh <= tt

此外，这里最后i>=k-1，假如说k=3的时候，那么就当i=2的时候，就要加入元素进去，因为数组是从0开始的

```
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        vector<int> res;
        vector<int> q(nums.size());

        int hh = 0, tt = -1;

        for (int i = 0; i < nums.size(); i++) {
            while (hh <= tt && i - k + 1 > q[hh]) {
                hh++;
            }

            while (hh <= tt && nums[i] >= nums[q[tt]]) {
                tt--;
            }

            q[++tt] = i;

            if (i >= k - 1) {
                res.push_back(nums[q[hh]]);
            }
        }

        return res;
    }
};
```

## [76.最小覆盖子串](https://leetcode.cn/problems/minimum-window-substring/)

有难度，很多小细节不好想

```
class Solution {
public:
    string minWindow(string s, string t) {
        vector<int> need(128, 0);

        // 记录下来需要匹配的字母每个字母各有多少个
        for (auto c : t) {
            need[c]++;
        }

        int cnt = t.size();
        int st = 0;
        int minLen = 0x3f3f3f3f;
        int l = 0; // 左边界

        for (int r = 0; r < s.size(); r++) {
            int index = s[r];

            if (need[index] > 0) {
                cnt--;
            }
            need[index]--; // 入窗口，-1

            while (cnt == 0) {
                if (minLen > r - l + 1) {
                    minLen = r - l + 1;
                    st = l;
                }

                need[s[l]]++; // 左边界右移

                if (need[s[l]] > 0) {
                    cnt++;
                }
                l++;
            }
        }

        if (minLen == 0x3f3f3f3f)
            return "";
        else
            return s.substr(st, minLen);
    }
};
```

其中：

- `first`：要反转区间的起始位置
- `last`：要反转区间结束位置的后一个位置
- 反转范围是 `[first, last)`

注意，这是一个**左闭右开区间**：

```

也就是：

- 包含 `first`
- 不包含 `last`

```
class Solution {

        int n = nums.size();

        k %= n;//主要是防止k>n

        reverse(nums.begin(), nums.end());

        reverse(nums.begin(), nums.begin() + k);

        reverse(nums.begin() + k, nums.end());
    }

class Solution {

        bool fist_row = false;
        bool fist_line = false;

        int n = matrix.size();
        int m = matrix[0].size();

        for(int i = 0; i < n; i ++ ){
            if(matrix[i][0] == 0){
                fist_line = true;
                break;
            }
        }

        for(int i = 0; i < m; i ++ ){
            if(matrix[0][i] == 0){
                fist_row = true;
                break;
            }
        }

        for(int i = 0; i < n; i ++ ){
            for(int j = 0; j < m; j ++ ){
                if(matrix[i][j] == 0){
                    matrix[i][0] = 0;
                    matrix[0][j] = 0;
                }
            }
        }

        for(int i = 1; i < n; i ++ ){
            for(int j = 1; j < m; j ++ ){
                if(matrix[i][0] == 0 || matrix[0][j] == 0){
                    matrix[i][j] = 0;
                }
            }
        }

        if(fist_line == true){
            for(int i = 0; i < n; i ++ ){
                matrix[i][0] = 0;
            }
        }

        if(fist_row == true){
            for(int i = 0; i < m; i ++ ){
                matrix[0][i] = 0;
            }
        }
    }
};

## [54. 螺旋矩阵](https://leetcode.cn/problems/spiral-matrix/)

```
class Solution {
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        int n = matrix.size();
        int m = matrix[0].size();

        int top = 0;
        int bottom = n - 1;
        int l = 0;
        int r = m - 1;

        vector<int> res;

        //先左到右，再上到下，再右到做，最后下到上
        while(top <= bottom && l <= r){
            for(int i = l; i <= r; i ++ ){
                res.push_back(matrix[top][i]);
            }
            top ++;

            for(int i = top; i <= bottom; i ++ ){
                res.push_back(matrix[i][r]);
            }
            r --;

            if(top <= bottom){
                for(int i = r; i >= l; i -- ){
                    res.push_back(matrix[bottom][i]);
                }
                bottom --;
            }

            if(l <= r){
                for(int i = bottom; i >= top; i -- ){
                    res.push_back(matrix[i][l]);
                }
                l ++;
            }
        }

        return res;
    }
};
```

## [240. 搜索二维矩阵 II](https://leetcode.cn/problems/search-a-2d-matrix-ii/)

最合适的方法是从**右上角**开始搜索。

假设当前位置是：

```
matrix[i][j]
```

有三种情况：

- 等于 `target`：直接返回 `true`
- 大于 `target`：这一列当前位置下面的数只会更大，所以向左移动
- 小于 `target`：这一行当前位置左边的数只会更小，所以向下移动

```
class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int n = matrix.size();
        int m = matrix[0].size();

        int i = 0;
        int j = m - 1;

        while(i < n && j >= 0){
            if(matrix[i][j] == target){
                return true;
            }
            else if(matrix[i][j] < target){
                i ++;
            }
            else{
                j --;
            }
        }

        return false;
    }
};
```

## [160. 相交链表](https://leetcode.cn/problems/intersection-of-two-linked-lists/)

两个指针分别从头开始走，走到结尾了，就从另外一条链表的头开始，一定会相遇的，相遇的部分若为：a+公共部分+b，那么a指针就是走的a+公共部分+b，b指针就是走的b+公共部分+a

两个指针分别从 `headA`、`headB` 出发：

- `pA` 走到链表 A 末尾后，改为从 `headB` 开始走。
- `pB` 走到链表 B 末尾后，改为从 `headA` 开始走。
- 最终两个指针会在相交节点相遇；如果不相交，会同时走到 `nullptr`。

```
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        ListNode* pa = headA;
        ListNode* pb = headB;

        while(pa != pb){
            if(pa == NULL){
                pa = headB;
            }
            else{
                pa = pa -> next;
            }

            if(pb == NULL){
                pb = headA;
            }
            else{
                pb = pb -> next;
            }
        }

        return pa;
    }
};
```

## [206. 反转链表](https://leetcode.cn/problems/reverse-linked-list/)

/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        ListNode* cur = head;
        ListNode* prev = nullptr;

        while(cur != nullptr){
            ListNode* next_node = cur -> next;

            cur -> next = prev;
            prev = cur;
            cur = next_node;
        }

        return prev;
    }
};
```

## [234. 回文链表](https://leetcode.cn/problems/palindrome-linked-list/)

这个题目是比较简单的，就是引用了链表倒置，步骤：

1. 快慢指针找中点
2. 中点之后倒置
3. 前一部分和后一部分一一对比，如果说有问题，返回false，如果没有问题，最后输出true

```
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverseList(ListNode* head){
        ListNode* cur = head;
        ListNode* pre = nullptr;

        while(cur != nullptr){
            ListNode* nextNode = cur -> next;

            cur -> next = pre;
            pre = cur;
            cur = nextNode;
        } 

        return pre;
    }
    bool isPalindrome(ListNode* head) {
        ListNode* slow = head;
        ListNode* fast = head;

        while(fast -> next != nullptr && fast -> next -> next != nullptr){
            fast = fast -> next -> next;
            slow = slow -> next;
        }

        ListNode* rest = reverseList(slow -> next);
        ListNode* st = head;

        while(rest != nullptr){
            if(rest -> val != st -> val){
                return false;
            }
            rest = rest -> next;
            st = st -> next;
        }

        return true;
    }
};
```

## [141. 环形链表](https://leetcode.cn/problems/linked-list-cycle/)

这里搞清了一个问题，就是fast != NULL && fast -> next != NULL与fast -> next != nullptr && fast -> next -> next != nullptr逻辑判断问题：
前者：slow指针会指向中间或者右半部分的第一个

后者：slow指针会指向中间或者左半部分的最后一个

```
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    bool hasCycle(ListNode *head) {

 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverseKGroup(ListNode* head, int k) {
        ListNode dummy(0, head);
        ListNode* prev = &dummy;

        while(1){
            ListNode* kth = prev;

            for(int i = 0; i < k && kth != nullptr; i ++ ){
                kth = kth -> next;
            }

            if(kth == nullptr) break;//不足k个节点

            ListNode* groupNext = kth -> next;//下一组的第一个节点保存下来

            //翻转当前链表
            ListNode* node = groupNext;
            ListNode* cur = prev -> next;

            while(cur != groupNext){
                ListNode* nextNode = cur -> next;
                cur -> next = node;
                node = cur;
                cur = nextNode;
            }

            //调转首尾指针
            ListNode* oldGroupHead = prev -> next;

            prev -> next = kth;

            prev = oldGroupHead;
        }

        return dummy.next;
    }
};

## [148. 排序链表](https://leetcode.cn/problems/sort-list/)

这道题就是先写一个合并的函数（之前写过），然后使用快慢指针（注意细节），将这个链表分成两部分，然后再分别排序，最后合并，里面有很多小细节需要注意

```
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    //合并
    ListNode* mergeList(ListNode* l1, ListNode* l2){
        ListNode dummy;
        ListNode* tail = &dummy;

        while(l1 != nullptr && l2 != nullptr){
            if(l1 -> val < l2 -> val){
                tail -> next = l1;
                l1 = l1 -> next;
            }
            else{
                tail -> next = l2;
                l2 = l2 -> next;
            }
            tail = tail -> next;
        }

        if(l1 != nullptr) tail -> next = l1;
        if(l2 != nullptr) tail -> next = l2;

        return dummy.next;
    }

    //先快慢指针拆两部分，分别排序之后再合并
    ListNode* sortList(ListNode* head) {
        if (head == nullptr || head->next == nullptr) {
            return head;
        }

        ListNode* slow = head;
        ListNode* fast = head -> next;//这块儿如果是head的话，会出现错误，例如1,2，起始时都不为空，然后slow指向2，fast指向null，然后切分，这样会一直循环下去

        while(fast != nullptr && fast -> next != nullptr){
            slow = slow -> next;
            fast = fast -> next -> next;
        }

        ListNode* rightHead = slow -> next;
        slow -> next = nullptr;

        ListNode* left = sortList(head);
        ListNode* right = sortList(rightHead);

        ListNode* res = mergeList(left, right);

        return res;
    }
};
```

## [146. LRU 缓存](https://leetcode.cn/problems/lru-cache/)

```
class LRUCache {
    int cap;
    list<pair<int, int>> cache;//保存键值对，list是stl容器
    unordered_map<int, list<pair<int, int>>::iterator> h;//根据key找到对于的cache
public:
    LRUCache(int capacity) {
        cap = capacity;
    }
    
    int get(int key) {
        if(h.find(key) == h.end()) return -1;//没找到对于的key

        auto t = h[key];
        int value = t -> second;

        cache.erase(t);
        cache.push_front({key, value});

        h[key] = cache.begin();

        return value;
    }
    
    void put(int key, int value) {
        //如果存在这个key
        if(h.find(key) != h.end()){
            auto t = h[key];

            cache.erase(t);
            cache.push_front({key, value});

            h[key] = cache.begin();

            return ;
        }

        //如果一开始不存在key，并且容量满了
        if(h.size() == cap){
            int oldLastKey = cache.back().first;

            cache.pop_back();
            h.erase(oldLastKey);
        }

        cache.push_front({key, value});
        h[key] = cache.begin();
    }
};

 * LRUCache* obj = new LRUCache(capacity);
 * int param_1 = obj->get(key);
 * obj->put(key,value);

## [230. 二叉搜索树中第 K 小的元素](https://leetcode.cn/problems/kth-smallest-element-in-a-bst/)

思路：中序遍历对应着从小到大的排列

一路向左走，遇到左孩子就压栈，走不通了就弹栈，每弹出来一个数就k--，若k不等于0，那么就压右孩子，压到走不通了再弹栈

 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;

class Solution {

    //有递归和堆栈两种方式实现
    //堆栈法：
    int kthSmallest(TreeNode* root, int k) {
       stack<TreeNode*> st;
       TreeNode* cur = root;

       while(cur != nullptr || st.size()){
            while(cur != nullptr){
                st.push(cur);
                cur = cur -> left;
            }

            cur = st.top();
            st.pop();

            k --;
            if(k == 0){
                return cur -> val;
            }

            cur = cur -> right;
       }

       return -1;
    }
};

## [105. 从前序与中序遍历序列构造二叉树](https://leetcode.cn/problems/construct-binary-tree-from-preorder-and-inorder-traversal/)

建树的方法是前序序列构造的，因为前序序列，根节点永远在前面，后面跟着的是左子树和右子树，通过中序序列，知道树的范围，如果l>r，那么就不递归建树了，当前中序区间没有根节点，如果在区间里面，因为前序前面说了，永远是根节点在前面，第一个节点永远是根节点，后面的区间在不断找左子树的根节点，右子树的根节点，然后练起来。总的来说，就是前序序列建树，因为根节点特性，比较好建树，而中序序列用来找子树区间

```
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    unordered_map<int, int> h;//保存中序遍历节点值和位置的关系
    int preorderIndex = 0;//先序遍历根节点在前面
    
    TreeNode* build(vector<int>& preorder, int l, int r){
        if(l > r) return nullptr;//中序遍历的序列是空的

        int rootValue = preorder[preorderIndex ++ ];

        TreeNode* root = new TreeNode(rootValue);

        int rooIndex = h[rootValue];

        root -> left = build(preorder, l, rooIndex - 1);
        root -> right = build(preorder, rooIndex + 1, r);

        return root;
    }
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        for(int i = 0; i < inorder.size(); i ++ ){
            h[inorder[i]] = i;
        }

        return build(preorder, 0, preorder.size() - 1);
    }
};
```

## [437. 路径总和 III](https://leetcode.cn/problems/path-sum-iii/)

count函数主要是递归计算从当前节点出发，能走到target的路径有几条

pathSum函数主要是递归模拟每个节点，能走到target的路径有几条

题目有点难度，可以多看几遍

因为第二个参数 `targetSum` 在 `count()` 里已经不再表示“题目给的目标值”，而是在表示 **“走到当前节点时，还差多少”**。这个值会随着递归不断做减法，可能远远超出 `int` 的范围。所以在count函数里面是long long类型

```
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    int count(TreeNode* node, long long target){
        if(node == nullptr) return 0;

        int num = 0;

        if(node -> val == target){
            num ++;
        }
		//这里不直接返回num，是因为下面可能会有正数负数，还可能有不同的路径
        num += count(node -> left, target - node -> val);
        num += count(node -> right, target - node -> val);

        return num;

    }
    int pathSum(TreeNode* root, int targetSum) {
        if(root == nullptr) return 0;

        int num = count(root, targetSum);

        num += pathSum(root -> left, targetSum);
        num += pathSum(root -> right, targetSum);

        return num;
    }
};
```

## [236. 二叉树的最近公共祖先](https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-tree/)

递归搜索左右子树，再根据左右子树返回的信息，判断当前节点是不是最近公共祖先。

递归有点恶心，快做吐了，又不好想

从下往上找：如果 `p`、`q` 分别出现在当前节点的左右两棵子树里，那么当前节点就是最近公共祖先。

```
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Solution {
public:
    //以 root 为根的子树中寻找 p 和 q。
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        if(root == NULL) return NULL;

        if(root == p || root == q) return root;//如果根节点为其中的一个节点，直接返回，对于题目中说的一个节点也可以是它自己的祖先，示例2

        TreeNode* l = lowestCommonAncestor(root -> left, p, q);
        TreeNode* r = lowestCommonAncestor(root -> right, p, q);

        if(l != NULL && r != NULL) return root;//p，q分别在左右子树

        if(l != NULL){
            return l;
        }//右子树为空，向上传递l结果

        return r;//左子树为空，向上传递r结果

    }
};
```

## [124. 二叉树中的最大路径和](https://leetcode.cn/problems/binary-tree-maximum-path-sum/)

这道题有点难，很难想到，主要的思路是每个节点都尝试作为一次路径的“最高点”，计算经过它的最大路径和；同时递归向父节点返回一条不能分叉的最大路径。

```
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    int ans = -1e5;
    int dfs(TreeNode* root){
        if(root == nullptr) return 0;

        int l = max(0, dfs(root -> left));
        int r = max(0, dfs(root -> right));//防止该路径的贡献是负数的情况

        int sum = l + r + root -> val;

        ans = max(ans, sum);

        return root -> val + max(l, r);
    }
    int maxPathSum(TreeNode* root) {
        dfs(root);

        return ans;
    }
};
```

## [994. 腐烂的橘子](https://leetcode.cn/problems/rotting-oranges/)

多源bfs，一定要注意是多源，不是单源，然后就是进行腐烂的时候，需要加条件判断，还是比较复杂的，容易忘条件，a了好几次才过

```
class Solution {
public:
    int orangesRotting(vector<vector<int>>& grid) {
        int n = grid.size();
        int m = grid[0].size();

        queue<pair<int, int>> q;

        int fresh = 0;
        int res = 0;

        for(int i = 0; i < n; i ++ ){
            for(int j = 0; j < m; j ++ ){
                if(grid[i][j] == 2) q.push({i, j});
                else if(grid[i][j] == 1) fresh ++;
            }
        }

        int dx[4] = {-1, 0, 1, 0}, dy[4] = {0, 1, 0, -1};

        while(q.size()){
            auto t = q.front();
            q.pop();
            bool rot = false;

            for(int i = 0; i < 4; i ++ ){
                int a = t.first + dx[i];
                int b = t.second + dy[i];

                if(a < 0 || a >= n || b < 0 || b >= m) continue;//越界

                if(grid[a][b] == 1){
                    grid[a][b] = 2;//将好橘子标记为坏橘子
                    fresh --;
                    rot = true;
                    q.push({a, b});
                }
            }
            if(rot == true) res ++;//本轮有橘子腐烂，时间+1
        }
        if(fresh == 0) return res;
        return -1;
    }
};

## [39. 组合总和](https://leetcode.cn/problems/combination-sum/)

思路：从当前位置开始枚举候选数字。因为同一个数字可以重复使用，递归时仍然传入当前下标 `i`；为了避免出现 `[2,3,2]`和 `[2,2,3]` 这种重复组合，只向后选择数字。

```
class Solution {
public:
    vector<vector<int>> res;
    vector<int> path;

    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        sort(candidates.begin(), candidates.end());

        backtrack(candidates, target, 0);

        return res;
    }

    void backtrack(vector<int>& candidates, int target, int u ){
        if(target == 0){
            res.push_back(path);
        }

        for(int i = u; i < candidates.size(); i ++ ){
            if(candidates[i] > target){
                break;
            }

            path.push_back(candidates[i]);

            backtrack(candidates, target - candidates[i], i);

            path.pop_back();
        }
    }
};

class Solution {

        backtrak(path, 0, 0, n);

            res.push_back(path);
            return ;

            backtrak(path, l + 1, r, n);
            path.pop_back();
        }

        if(r < l){
            path += ')';
            backtrak(path, l, r + 1, n);
            path.pop_back();
        }
    }

        m = board[0].size();

            for(int j = 0; j < m; j ++ ){
                dfs(board, word, 0, i, j);
            }
        }

        return f;
    }

    void dfs(vector<vector<char>>& board, string word, int u, int x, int y){
        if(board[x][y] == word[u]){
            if(u == word.size() - 1){
            f = true;
            return ;
        }

            int t = board[x][y];
            board[x][y] = '#';
            for(int i = 0; i < 4; i ++ ){
                int a = x + dx[i];
                int b = y + dy[i];

                if(a >= 0 && a < n && b >= 0 && b < m){
                    dfs(board, word, u + 1, a, b);
                }
            }
            board[x][y] = t;
        }
        
    }

## [131. 分割回文串](https://leetcode.cn/problems/palindrome-partitioning/)

从字符串下标 `start` 开始，枚举当前子串的结束位置 `end`：

- 如果 `s[start...end]` 是回文串，就加入当前路径。
- 然后从 `end + 1` 继续切分。
- 当 `start == s.size()` 时，说明整个字符串已经切分完成，记录答案。

    vector<vector<string>> res;
    vector<string> path;

    vector<vector<string>> partition(string s) {
        dfs(s, 0);
        return res;
    }

    //判断是不是回文串
    bool is_huiwen(string& s, int l, int r){
        while(l < r){
            if(s[l] != s[r]){
                return false;
            }
            l ++, r --;
        }
        return true;
    }

    //DFS
    void dfs(string& s, int st){
        if(st == s.size()){
            res.push_back(path);//走到末尾，添加一条路径
        }

        for(int ed = st; ed < s.size(); ed ++ ){
            if(!is_huiwen(s, st, ed)) continue;

            path.push_back(s.substr(st, ed - st + 1));

            dfs(s, ed + 1);

            path.pop_back();//回溯
        }
    }
};

## [4. 寻找两个正序数组的中位数](https://leetcode.cn/problems/median-of-two-sorted-arrays/)

这道题有点难度，但是你找中位数，可以看成找第k小的数，这个k是中间位数，比如说你有5位数，就找5 / 2 + 1 = 3小的数

```
class Solution {
public:
    double findMedianSortedArrays(vector<int>& nums1, vector<int>& nums2) {
        int n = nums1.size();
        int m = nums2.size();

        int sum = n + m;

        if(sum % 2 == 1) return getKth(nums1, nums2, sum / 2 + 1);

        int l = getKth(nums1, nums2, sum / 2);
        int r = getKth(nums1, nums2, sum / 2 + 1);

        return (l + r) / 2.0;
    }

    //找第k小的数
    int getKth(vector<int>& nums1, vector<int>& nums2, int k){
        int n = nums1.size();
        int m = nums2.size();

        int i = 0;//s1指针
        int j = 0;//s2指针

        while(true){
            //数组判空要放在k == 1前面
            if(i == n) return nums2[j + k - 1];//第一个数组走完
            if(j == m) return nums1[i + k - 1];//第二个数组走完，答案只可能在第一个数组里面

            if(k == 1) return min(nums1[i], nums2[j]);//表面是查找第一个数，返回最小的即可

            int half = k / 2;
            
            //防止越界
            int x = min(i + half, n) - 1;
            int y = min(j + half, m) - 1;

            //不可能在[i, ... , x]里面
            if(nums1[x] <= nums2[y]){
                k -= x - i + 1;
                i = x + 1;
            }
            else{
                k -= y - j + 1;
                j = y + 1;
            }

        }  
        
    }
};
```

dp[n] = dp[n - 1] + dp[n - 2]

```
class Solution {
public:
    int climbStairs(int n) {
        if(n <= 2) return n;

        int a = 1, b = 2;

        for(int i = 3; i <= n; i ++ ){
            int c = a + b;
            a = b;
            b = c;
        }

        return b;
    }
};
```

## [152. 乘积最大子数组](https://leetcode.cn/problems/maximum-product-subarray/)

对于当前 `nums[i]`，以它结尾的连续子数组只有三种来源：

```
1. 从 nums[i] 自己重新开始
2. 接在之前的最大乘积后面：oldmax * nums[i]
3. 接在之前的最小乘积后面：oldmin * nums[i]
```

```
class Solution {
public:
    //这道题有一个坑点，就是最小负数乘以当前的负数，可能会变成最大值
    //所以在进行处理的时候，不仅仅要保留最大值，也要保留最小值
    int maxProduct(vector<int>& nums) {
        int n = nums.size();

        int maxNum = nums[0];
        int minNum = nums[0];
        int res = nums[0];

        for(int i = 1; i < n; i ++ ){
            int num = nums[i];

            int oldMax = maxNum;
            int oldMin = minNum;

            maxNum = max(num, max(oldMax * num, oldMin * num));
            minNum = min(num, min(oldMax * num, oldMin * num));

            res = max(res, maxNum);
        }

        return res;
    }
};
```

## [416. 分割等和子集](https://leetcode.cn/problems/partition-equal-subset-sum/)

```
class Solution {
public:
    //01背包问题：
    //0/1 背包倒序遍历，因为每个数字只能选一次。
    bool canPartition(vector<int>& nums) {
        int sum = 0;
        int n = nums.size();

        for(int i = 0; i < nums.size(); i ++ ){
            sum += nums[i];
        }

        if(sum % 2 == 1) return false;

        //如果是偶数，那么如果能凑出一半，剩下的数自动构成另外一半
        int target = sum / 2;

        vector<bool> dp(target + 1, false);
        dp[0] = true;

        for(int i = 0; i < n; i ++ ){
            for(int j = target; j >= nums[i]; j --){
                dp[j] = dp[j] || dp[j - nums[i]];
            }
        }

        return dp[target];
    }
};
```

这道题的动态规划方程：

    int uniquePaths(int m, int n) {
        vector<vector<int>> dp(m, vector<int>(n, 1));

                dp[i][j] = dp[i - 1][j] + dp[i][j - 1];
            }
        }

        return dp[m - 1][n - 1];
    }

## [64. 最小路径和](https://leetcode.cn/problems/minimum-path-sum/)

和上道题是类似的，都需要初始化第一行第一列，然后处理中间节点，但是动态规划方程需要更改

定义：

```
dp[i][j]
```

表示：

> 从左上角 `(0,0)` 走到 `(i,j)` 的最小路径和。

机器人只能向右或向下，所以到达 `(i,j)` 之前，只可能位于：

```
上方：(i - 1, j)
左方：(i, j - 1)
```

因此状态转移是：

```
dp[i][j] = min(dp[i - 1][j], dp[i][j - 1]) + grid[i][j];
```

含义是：

> 选择到达上方和左方的两条路径中较小的，再加上当前格子的数字。

```
class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int n = grid.size();
        int m = grid[0].size();

        vector<vector<int>> dp(n, vector<int>(m, 0));

        dp[0][0] = grid[0][0];

        // 初始化第一行
        for (int i = 1; i < m; i++) {
            dp[0][i] = dp[0][i - 1] + grid[0][i];
        }

        // 初始化第一列
        for (int i = 1; i < n; i++) {
            dp[i][0] = dp[i - 1][0] + grid[i][0];
        }

        for (int i = 1; i < n; i++) {
            for (int j = 1; j < m; j++) {
                dp[i][j] = min(dp[i - 1][j], dp[i][j - 1]) + grid[i][j];
            }
        }

        return dp[n - 1][m - 1];
    }
};
```

## [5. 最长回文子串](https://leetcode.cn/problems/longest-palindromic-substring/)

**中心扩展法**，比二维动态规划更容易理解。

回文串的特点是：

> 从中心向左右两边扩展时，左右字符始终相等。

```
class Solution {
public:
    int st = 0;
    int res = 0;

    // 从一个点或者一对点向两边扩展
    void expand(string& s, int l, int r) {
        while (l >= 0 && r < s.size() && s[l] == s[r]) {
            l--;
            r++;
        }

        // 退出循环的时候，l多减了1， r多加了1
        int len = (r - 1) - (l + 1) + 1;

        if (len > res) {
            res = len;
            st = l + 1;
        }
    }

    string longestPalindrome(string s) {
        for (int i = 0; i < s.size(); i++) {
            expand(s, i, i);
            expand(s, i, i + 1);
        }

        return s.substr(st, res);
    }
};
```

            expand(s, ans, i, i, cnt);
            expand(s, ans, i, i + 1, cnt);

            return ;
        }

        while(l >= 0 && r < s.size()){
            if(s[l] == s[r]){
                l --, r ++;
                int len = r - l - 1;

                if(len > cnt){
                    cnt = len;
                    ans = s.substr(l + 1, len);
                }
            }
            else{
                break;
            }
        }
    }
};

class Solution {

            }
        }

## [31. 下一个排列](https://leetcode.cn/problems/next-permutation/)

这道题是固定的套路，记下来就好了

核心步骤只有 3 个：

1. 从右往左找第一个“下降点” `i`
2. 从右往左找第一个比 `nums[i]` 大的数 `nums[j]`
3. 交换后，把 `i` 后面的部分反转

```
class Solution {
public:
    void nextPermutation(vector<int>& nums) {
        int n = nums.size();

        int i = n - 2;

        while(i >=0 && nums[i] >= nums[i + 1]){
            i --;
        }

        //如果找到了i
        if(i >= 0){
            int j = n - 1;
            while(j > i && nums[i] >= nums[j]){
                j --;
            }
            swap(nums[i], nums[j]);
        }

        reverse(nums.begin() + i + 1, nums.end());
    }
};
```

## [287. 寻找重复数](https://leetcode.cn/problems/find-the-duplicate-number/)

这道题可以看成找环的入口，把静态数组看成指针

例如：

```
nums = [1, 3, 4, 2, 2]
```

指向关系为：

```
下标 0 → nums[0] = 1
下标 1 → nums[1] = 3
下标 3 → nums[3] = 2
下标 2 → nums[2] = 4
下标 4 → nums[4] = 2
```

所以从下标 `0` 出发：

```
0 → 1 → 3 → 2 → 4 → 2 → 4 → 2 ...
```

找环的入口即可，代码不难

```
class Solution {
public:
    int findDuplicate(vector<int>& nums) {
        int slow = 0;
        int fast = 0;

        // 找环的相遇点
        do {
            slow = nums[slow];
            fast = nums[nums[fast]];
        } while (slow != fast);

        // 找环的起点
        slow = 0;
        while (slow != fast) {
            slow = nums[slow];
            fast = nums[fast];
        }

        return slow;
    }
};
```
